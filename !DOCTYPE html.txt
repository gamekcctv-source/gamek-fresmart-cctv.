        function calculateDiffInfo(finalVal, sysVal) {
            const hasFinal = finalVal !== '' && finalVal !== null && finalVal !== undefined && !isNaN(Number(finalVal));
            const hasSys = sysVal !== '' && sysVal !== null && sysVal !== undefined && !isNaN(Number(sysVal));

            if (!hasFinal || !hasSys) {
                return { status: 'pending', text: '—', value: null };
            }

            const diff = Number(finalVal) - Number(sysVal);
            if (diff === 0) {
                return { status: 'ok', text: '0 (OK)', value: 0 };
            }
            const formatted = diff > 0 ? `+${diff}` : `${diff}`;
            return { status: 'diff', text: formatted, value: diff };
        }

        function updateStats() {
            let okCount = 0;
            let diffCount = 0;
            let auditedCount = 0;
            const totalItems = appState.products.length;

            appState.products.forEach(p => {
                const entry = appState.entries[p.id] || {};
                const info = calculateDiffInfo(entry.final, entry.sistema);
                if (info.status === 'ok') {
                    okCount++;
                    auditedCount++;
                } else if (info.status === 'diff') {
                    diffCount++;
                    auditedCount++;
                }
            });

            const statOkEl = document.getElementById('stat-ok');
            const statDiffEl = document.getElementById('stat-diff');
            const progressBarEl = document.getElementById('audit-progress-bar');
            const progressTextEl = document.getElementById('audit-progress-text');

            if (statOkEl) statOkEl.textContent = okCount;
            if (statDiffEl) statDiffEl.textContent = diffCount;

            const pct = totalItems > 0 ? Math.round((auditedCount / totalItems) * 100) : 0;
            if (progressBarEl) progressBarEl.style.width = `${pct}%`;
            if (progressTextEl) progressTextEl.textContent = `${auditedCount}/${totalItems} (${pct}%)`;
        }

        function renderTable() {
            const tbody = document.getElementById('inventory-tbody');
            const searchInput = document.getElementById('search-input');
            const clearBtn = document.getElementById('clear-search-btn');
            const query = searchInput.value.trim().toLowerCase();
            const catFilter = document.getElementById('category-filter').value;

            if (query) {
                clearBtn.classList.remove('hidden');
            } else {
                clearBtn.classList.add('hidden');
            }

            const filtered = appState.products.filter(p => {
                const matchesCat = catFilter === 'ALL' || p.category === catFilter;
                const matchesQuery = !query ||
                    p.code.toLowerCase().includes(query) ||
                    p.description.toLowerCase().includes(query);
                return matchesCat && matchesQuery;
            });

            if (filtered.length === 0) {
                tbody.innerHTML = `
                    <tr>
                        <td colspan="10" class="py-10 text-center text-slate-500">
                            Nenhum produto encontrado para o filtro aplicado.
                        </td>
                    </tr>
                `;
                updateStats();
                return;
            }

            let html = '';
            let lastCategory = null;

            filtered.forEach((prod, idx) => {
                if (prod.category !== lastCategory) {
                    lastCategory = prod.category;
                    html += `
                        <tr class="category-row bg-slate-800 text-amber-300">
                            <td colspan="10" class="py-1.5 px-3 text-xs font-bold uppercase tracking-wider">
                                ${prod.category}
                            </td>
                        </tr>
                    `;
                }

                const e = appState.entries[prod.id] || {};
                const diffInfo = calculateDiffInfo(e.final, e.sistema);
                const diffClasses = diffInfo.status === 'ok'

        /**
         * Renderiza a folha inteira em Canvas usando html2canvas (scale: 2)
         */
        async function captureFullSheetCanvas() {
            if (typeof html2canvas === 'undefined') {
                showToast('Biblioteca de captura ainda carregando. Tente novamente em instantes.', 'warning');
                return null;
            }

            // Reseta filtros temporariamente se houver busca ativa para garantir folha completa
            const searchInput = document.getElementById('search-input');
            const catFilter = document.getElementById('category-filter');
            const prevSearch = searchInput.value;
            const prevCat = catFilter.value;

            if (prevSearch || prevCat !== 'ALL') {
                searchInput.value = '';
                catFilter.value = 'ALL';
            }

            // Re-renderiza garantindo textos estáticos atualizados para a captura limpa
            renderTable();
            document.getElementById('sig-display-cctv').textContent = appState.signatures.cctv?.text || '';
            document.getElementById('sig-display-manager').textContent = appState.signatures.manager?.text || '';

            document.body.classList.add('capturing-sheet');
            await new Promise(resolve => setTimeout(resolve, 150));

            try {
                const captureTarget = document.getElementById('capture-area');
                const canvas = await html2canvas(captureTarget, {
                    scale: 2,
                    useCORS: true,
                    backgroundColor: '#fefce8',
                    logging: false,
                    windowWidth: 1280
                });

                lastCapturedCanvas = canvas;
                lastCapturedDataUrl = canvas.toDataURL('image/png');
                return canvas;
            } finally {
                document.body.classList.remove('capturing-sheet');
                if (prevSearch || prevCat !== 'ALL') {
                    searchInput.value = prevSearch;
                    catFilter.value = prevCat;
                    renderTable();
                }
            }
        }

        /**
         * 2. Gerar e Baixar PDF Direto (A4 Paisagem) + Opção de Envio
         */
        async function buildPDFDocument() {
            const canvas = await captureFullSheetCanvas();
            if (!canvas) return null;

            const { jsPDF } = window.jspdf;
            const pdf = new jsPDF({
                orientation: 'landscape',
                unit: 'mm',
                format: 'a4'
            });

            const pageWidth = pdf.internal.pageSize.getWidth();
            const pageHeight = pdf.internal.pageSize.getHeight();

            // Fundo amarelo suave fiel à folha física em toda a página A4
            pdf.setFillColor(254, 252, 232);
            pdf.rect(0, 0, pageWidth, pageHeight, 'F');

            const margin = 5;
            const usableWidth = pageWidth - margin * 2;
            const usableHeight = pageHeight - margin * 2;

            const imgData = canvas.toDataURL('image/png');
            const imgRatio = canvas.width / canvas.height;

            let renderWidth = usableWidth;
            let renderHeight = renderWidth / imgRatio;

            if (renderHeight > usableHeight) {
                renderHeight = usableHeight;
                renderWidth = renderHeight * imgRatio;
            }

            const offsetX = (pageWidth - renderWidth) / 2;
            const offsetY = (pageHeight - renderHeight) / 2;

            pdf.addImage(imgData, 'PNG', offsetX, offsetY, renderWidth, renderHeight, undefined, 'FAST');
            return pdf;
        }

            }
        }

        /**
         * 3. Compartilhamento Rápido Estilo Wooqer (Web Share API + WhatsApp / Facebook)
         */
        function buildOperationalSummary() {
            let okCount = 0;
            let diffItems = [];

            appState.products.forEach(prod => {
                const entry = appState.entries[prod.id] || {};
                const info = calculateDiffInfo(entry.final, entry.sistema);
                if (info.status === 'ok') okCount++;
                if (info.status === 'diff') {
                    diffItems.push(`• ${prod.code} (${prod.description}): Dif. ${info.text}${entry.obs ? ` [Obs: ${entry.obs}]` : ''}`);
                }
            });

            const lines = [
                `📋 *CONTROLE DE INVENTÁRIO DIÁRIO - CCTV / LOJA*`,
                `📅 *Data:* ${appState.date}`,
                appState.storeUnit ? `🏪 *Loja/Unidade:* ${appState.storeUnit}` : null,
                `🎥 *Operador CCTV:* ${appState.cctvOperator}`,
                `👔 *Gerente:* ${appState.storeManager}`,
                `✅ *Itens Conferidos OK:* ${okCount}`,
                `⚠️ *Divergências:* ${diffItems.length}`
            ].filter(Boolean);

            if (diffItems.length > 0) {
                lines.push('', '*Detalhes das Divergências:*', ...diffItems.slice(0, 10));
                if (diffItems.length > 10) {
                    lines.push(`...e mais ${diffItems.length - 10} item(ns).`);
                }
            }

            lines.push('', '📎 _Anexo: Folha de Inventário assinada (PNG/PDF)._');
            return lines.join('\n');
        }

        async function attemptDirectFileShare(canvas, summaryText) {
            if (!navigator.share || !navigator.canShare) return false;

            return new Promise((resolve) => {
                canvas.toBlob(async (blob) => {
                    if (!blob) {
                        resolve(false);
                        return;
                    }
                    const file = new File([blob], `Inventario_CCTV_${appState.date}.png`, { type: 'image/png' });
                    const shareData = {
                        title: `Inventário Diário CCTV - ${appState.date}`,
                        text: summaryText,
                        files: [file]
                    };

                    if (navigator.canShare(shareData)) {
                        try {
                            await navigator.share(shareData);
                            showToast('Folha compartilhada com sucesso!');
                            resolve(true);
                        } catch (shareErr) {
                            resolve(false);
                        }
                    } else {
                        resolve(false);
                    }
                }, 'image/png');
            });
        }

        async function triggerNativeShare(format = 'image') {
            if (!checkMandatorySignatures()) return;
            const summaryText = document.getElementById('share-summary-text').value || buildOperationalSummary();

            if (format === 'pdf') {
                if (!window.jspdf || !window.jspdf.jsPDF) {
                    showToast('Biblioteca PDF indisponível.', 'warning');
                    return;
                }
                toggleLoadingOverlay(true, 'Preparando PDF para envio...', 'Gerando arquivo PDF da folha assinada.');
                try {
                    const pdf = await buildPDFDocument();
                    if (!pdf) return;
                    const pdfBlob = pdf.output('blob');
                    const pdfFile = new File([pdfBlob], `Inventario_CCTV_${appState.date}.pdf`, { type: 'application/pdf' });

                    if (navigator.share && navigator.canShare && navigator.canShare({ files: [pdfFile] })) {
                        await navigator.share({
                            title: `Inventário Diário CCTV (PDF) - ${appState.date}`,
                            text: summaryText,
                            files: [pdfFile]
                        });
                        showToast('PDF compartilhado com sucesso!');
                    } else {
                        pdf.save(`Inventario_CCTV_${appState.date}.pdf`);
                        showToast('PDF baixado! Anexe o arquivo diretamente no WhatsApp ou app desejado.', 'warning');
                    }
                } finally {
                    toggleLoadingOverlay(false);
                }
                return;
            }

            if (!lastCapturedCanvas) return;
            const ok = await attemptDirectFileShare(lastCapturedCanvas, summaryText);
            if (!ok) {
                downloadLastScreenshot();
                showToast('Compartilhamento direto indisponível neste navegador. A imagem PNG foi baixada para anexar manualmente!', 'warning');
            }
        }

                const input = document.getElementById(`sig-input-${role}`);
                if (!appState.signatures[role]) {
                    appState.signatures[role] = { text: '', stampedAt: '' };
                }
                input.value = appState.signatures[role].text || '';

                input.addEventListener('input', (e) => {
                    const val = e.target.value;
                    appState.signatures[role].text = val;

                    if (val.trim() && !appState.signatures[role].stampedAt) {
                        const now = new Date();
                        const timeStr = now.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
                        const dateStr = now.toLocaleDateString('pt-BR');
                        const stamp = `✔ Assinado em ${dateStr} às ${timeStr}`;
                        appState.signatures[role].stampedAt = stamp;
                        document.getElementById(`sig-time-${role}`).textContent = stamp;
                    } else if (!val.trim()) {
                        appState.signatures[role].stampedAt = '';
                        document.getElementById(`sig-time-${role}`).textContent = '';
                    }

                    updateSignaturesUI();
                    saveState();
                });
            });
        }

        function clearSignature(role) {
            const input = document.getElementById(`sig-input-${role}`);
            if (input) input.value = '';
            appState.signatures[role] = { text: '', stampedAt: '' };
            document.getElementById(`sig-time-${role}`).textContent = '';
            updateSignaturesUI();
            saveState();
        }

        function stampValidation(role) {
            const text = (appState.signatures[role]?.text || '').trim();
            if (!text) {
                const input = document.getElementById(`sig-input-${role}`);
                if (input) input.focus();
                showToast('Digite a assinatura sobre a linha antes de confirmar!', 'warning');
                return;
            }
            const now = new Date();
            const timeStr = now.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
            const dateStr = now.toLocaleDateString('pt-BR');
            const stamp = `✔ Confirmado em ${dateStr} às ${timeStr}`;

            appState.signatures[role].stampedAt = stamp;
            document.getElementById(`sig-time-${role}`).textContent = stamp;
            updateSignaturesUI();
            saveState();
            showToast(role === 'cctv' ? 'Assinatura do Operador CCTV confirmada!' : 'Assinatura do Gerente confirmada!');
        }

        /**
         * Exporta os dados completos para planilha CSV compatível com Excel
         */
        function exportToCSV() {
            const headers = [
                'Categoria',
                'Codigo',
                'Descricao do Produto',
                'Qtd Inicial Loja',
                '08:00h',
                '14:00h',
                '20:00h',
                'Qtd Final Loja',
                'Qtd Sistema',
                'Diferenca',
                'Observacao'
            ];

            const rows = [
                [`CONTROLE DE INVENTARIO DIARIO - DATA: ${appState.date} | LOJA: ${appState.storeUnit || 'N/A'} | OPERADOR CCTV: ${appState.cctvOperator || 'N/A'} | GERENTE: ${appState.storeManager || 'N/A'}`],
                headers
            ];

            appState.products.forEach(prod => {
                const e = appState.entries[prod.id] || {};
                const hasFinal = e.final !== '' && e.final !== undefined;
                const hasSys = e.sistema !== '' && e.sistema !== undefined;
                const diff = (hasFinal && hasSys) ? (Number(e.final) - Number(e.sistema)) : '';

                rows.push([
                    `"${prod.category}"`,
                    `"${prod.code}"`,
                    `"${prod.description.replace(/"/g, '""')}"`,
                    e.inicial ?? '',
                    e.h08 ?? '',
                    e.h14 ?? '',
                    e.h20 ?? '',
                    e.final ?? '',
                    e.sistema ?? '',
                    diff,
                    `"${(e.obs ?? '').replace(/"/g, '""')}"`
                ]);
            });

            // BOM UTF-8 para abrir com acentuação perfeita no Excel
            const csvContent = '\uFEFF' + rows.map(r => r.join(';')).join('\n');
            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);

            const link = document.createElement('a');
            link.href = url;
            link.setAttribute('download', `Inventario_CCTV_Loja_${appState.date}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            URL.revokeObjectURL(url);

            showToast('Arquivo CSV exportado com sucesso!');
        }

        /**
         * Sincroniza inputs do cabeçalho com estado e rodapé de assinaturas
         */
        function bindHeaderInputs() {
            const dateInput = document.getElementById('shift-date');
            const storeInput = document.getElementById('store-unit');
            const cctvInput = document.getElementById('cctv-operator');
            const managerInput = document.getElementById('store-manager');

            dateInput.value = appState.date;
            storeInput.value = appState.storeUnit || '';
            cctvInput.value = appState.cctvOperator;
            managerInput.value = appState.storeManager;

            document.getElementById('sig-name-cctv').textContent = appState.cctvOperator || '______________________';
            document.getElementById('sig-name-manager').textContent = appState.storeManager || '______________________';

            if (appState.signatures.cctv?.stampedAt) {
                document.getElementById('sig-time-cctv').textContent = appState.signatures.cctv.stampedAt;
            }
            if (appState.signatures.manager?.stampedAt) {
                document.getElementById('sig-time-manager').textContent = appState.signatures.manager.stampedAt;
            }

            dateInput.addEventListener('change', (e) => {
                appState.date = e.target.value;
                saveState();
            });

            storeInput.addEventListener('input', (e) => {
                appState.storeUnit = e.target.value;
                saveState();
            });

            cctvInput.addEventListener('input', (e) => {
                appState.cctvOperator = e.target.value;
                document.getElementById('sig-name-cctv').textContent = e.target.value || '______________________';
                saveState();
            });

            managerInput.addEventListener('input', (e) => {
                appState.storeManager = e.target.value;
                document.getElementById('sig-name-manager').textContent = e.target.value || '______________________';
                saveState();
            });

            document.getElementById('search-input').addEventListener('input', renderTable);
            document.getElementById('category-filter').addEventListener('change', renderTable);
        }
